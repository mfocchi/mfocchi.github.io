(function () {
  var bar = document.getElementById('stu-tabbar');  
  var panels = document.querySelectorAll('.stu-panel');  
  if (!bar || !panels.length) { return; }  
  document.documentElement.classList.add('stu-js');  
  bar.hidden = false;  
  function show(key, push) {  
    var found = false;  
    for (var i = 0; i < panels.length; i++) {  
      var on = panels[i].id === 'stu-' + key;  
      panels[i].classList.toggle('active', on);  
      found = found || on;  
    }  
    if (!found) { return show(panels[0].id.replace('stu-', ''), false); }  
    var btns = bar.querySelectorAll('button');  
    for (var j = 0; j < btns.length; j++) {  
      btns[j].setAttribute('aria-selected', btns[j].getAttribute('data-tab') === key ? 'true' : 'false');  
    }  
    if (push && history.replaceState) { history.replaceState(null, '', '#' + key); }  
  }  
  bar.addEventListener('click', function (e) {  
    var b = e.target.closest('button');  
    if (b) { show(b.getAttribute('data-tab'), true); }  
  });  
  show(location.hash.replace('#', '') || 'phd', false);
})();
</script>
